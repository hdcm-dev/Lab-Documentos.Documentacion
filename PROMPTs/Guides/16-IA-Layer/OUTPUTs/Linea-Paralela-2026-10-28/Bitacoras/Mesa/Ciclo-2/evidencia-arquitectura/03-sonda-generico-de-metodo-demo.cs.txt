using Microsoft.Extensions.AI;
using Microsoft.Extensions.DependencyInjection;
using MovilidadUrbana.Asistente.Tests.Comun;

namespace SondaRojo.Application
{
    // Un tipo "del núcleo" que nombra ChatMessage sólo como argumento genérico de un método.
    public static class RegistroConMensaje
    {
        public static IServiceCollection Registrar(IServiceCollection s) => s.AddSingleton<ChatMessage>();
    }
}

namespace MovilidadUrbana.Asistente.Tests.Cableado
{
    [TestFixture, Category("Sonda")]
    public sealed class SondaGenericoDeMetodo
    {
        [Test]
        public void Argumento_generico_de_metodo()
        {
            var v = ReglaDeCapas.Violaciones([typeof(SondaRojo.Application.RegistroConMensaje)], ArquitecturaTests.Prohibido);
            TestContext.Out.WriteLine($"## Genérico de método: {v.Count} hallazgos");
            foreach (var x in v) TestContext.Out.WriteLine("   " + x);
            Assert.Pass();
        }
    }
}
