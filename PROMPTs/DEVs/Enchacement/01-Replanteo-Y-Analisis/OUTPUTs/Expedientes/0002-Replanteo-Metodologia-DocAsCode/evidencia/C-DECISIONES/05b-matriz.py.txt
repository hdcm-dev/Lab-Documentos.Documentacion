#!/usr/bin/env python3
"""Matriz de decisión C-DECISIONES: puntaje ponderado, sensibilidad ±10 %, barrido 0,10–0,40 y punto de inversión.
Pesos y escalas: evidencia/C-DECISIONES/01-pesos-asentados.md (asentados antes de puntuar)."""
import itertools, json, sys

W = {"K1": 0.25, "K2": 0.25, "K3": 0.25, "K4": 0.25}
K = list(W)

# Q -> {alternativa: (K1, K2, K3, K4)}; las observables de cada celda están en el informe 015 §5.
M = {
 "Q03": {"A1 única mismo repo": (3,2,4,2), "A2 única dedicado": (3,3,2,1),

         "P3 2651x completa + DITA + 26512": (1,2,3,4)},
 "Q46": {"No comprar; índice rotulado": (5,5,4,3), "Comprar ahora": (4,3,4,5),
         "Comprar al ocurrir el evento": (4,4,4,4)},
 "Q33": {"N1 migrar a 13.22 antes": (4,3,4,4), "N2 correr sobre 13.16": (5,5,2,2),
         "N3 migración selectiva": (4,3,3,1)},
}

def score(v, w): return sum(w[k]*x for k, x in zip(K, v))

def ranking(alts, w):
    return sorted(((score(v, w), a) for a, v in alts.items()), reverse=True)

def renorm(k, f):
    w = dict(W); w[k] = W[k]*f
    s = sum(w.values()); return {x: y/s for x, y in w.items()}

out = {}
for q, alts in M.items():
    r = ranking(alts, W)
    win, margin = r[0][1], r[0][0]-r[1][0]
    flips = []
    for k in K:
        for f in (0.9, 1.1):
            rw = ranking(alts, renorm(k, f))
            if rw[0][1] != win: flips.append(f"{k}x{f}->{rw[0][1]}")
    sweep = []
    for k in K:
        for i in range(10, 41):
            p = i/100; rest = (1-p)/3
            w = {x: (p if x == k else rest) for x in K}
            rw = ranking(alts, w)
            if rw[0][1] != win: sweep.append((k, p, rw[0][1])); break
    # punto de inversión: cuántos puntos debe bajar una sola celda del ganador para perder (peso 0,25)
    inv = round(margin/0.25, 2)
    firm = margin >= 0.30 - 1e-9 and not flips
    out[q] = dict(ranking=[(round(s, 2), a) for s, a in r], ganador=win, margen=round(margin, 2),
                  inversiones_10pct=flips, barrido_inversion=sweep, puntos_de_celda_para_invertir=inv,
                  decision="FIRME" if firm else "CON REVISION")

json.dump(out, sys.stdout, ensure_ascii=False, indent=1)
