def score(s, w):
    return round(sum(a * b for a, b in zip(s, w)), 4)


def winner(alts, w):
    ok = [(n, score(s, w)) for n, s, f in alts if f is None]
    ok.sort(key=lambda x: -x[1])
    return ok


def variants():
    out = {}
    for i in range(len(W)):
        for sign in (+1, -1):
            w = list(W)
            w[i] = W[i] * (1 + 0.10 * sign)
            t = sum(w)
            out[f"{CRIT[i].split()[0]}{'+' if sign > 0 else '-'}10%"] = [x / t for x in w]
    out["iguales"] = [1 / 7] * 7
    out["solo_TP36"] = [1 / 3, 1 / 3, 1 / 3, 0, 0, 0, 0]
    return out


res = {}
V = variants()
for d, alts in D.items():
    base = winner(alts, W)
    top, second = base[0], base[1]
    margin = round(top[1] - second[1], 4)
    flips = []
    for k, w in V.items():
        if winner(alts, w)[0][0] != top[0]:
            flips.append(k)
    pm10 = [f for f in flips if f.endswith("%")]
    estado = "FIRME" if margin >= 0.30 and not pm10 else "CON REVISIÓN"
    res[d] = {
        "puntajes": {n: score(s, W) for n, s, f in alts},
        "excluidas": {n: f for n, s, f in alts if f},
        "gana": top[0], "segunda": second[0], "margen": margin,
        "invierte_en": flips, "estado": estado,
    }

print("pesos:", dict(zip(CRIT, W)), "suma", round(sum(W), 4))
for d, r in res.items():
    print(f"\n{d}  {r['estado']}  margen {r['margen']:.2f}")
    for n, p in sorted(r["puntajes"].items(), key=lambda x: -x[1]):
        ex = f"  [excluida: {r['excluidas'][n]}]" if n in r["excluidas"] else ""
        print(f"  {p:.2f}  {n}{ex}")
    print("  se invierte con:", ", ".join(r["invierte_en"]) or "ninguna variante")
firmes = [d for d, r in res.items() if r["estado"] == "FIRME"]
rev = [d for d, r in res.items() if r["estado"] != "FIRME"]
print("\nFIRME:", " ".join(firmes))
print("CON REVISIÓN:", " ".join(rev))
with open("salida_a01.json", "w", encoding="utf-8") as f:
    json.dump(res, f, ensure_ascii=False, indent=1)
